# ============================================================
#  Mandi Minutes — Prerequisites & Requirements
#  (This is a Node.js / React project, not a Python project.
#   This file documents the tools you need to run it.)
# ============================================================


# ── Runtime ──────────────────────────────────────────────────
Node.js         >= 18.0.0    https://nodejs.org/en/download
npm             >= 9.0.0     (bundled with Node.js)
# OR use one of these Node version managers instead:
#   nvm (Linux/Mac)   https://github.com/nvm-sh/nvm
#   nvm-windows       https://github.com/coreybutler/nvm-windows
#   fnm (cross-platform, fast) https://github.com/Schniz/fnm


# ── Source Control ───────────────────────────────────────────
Git             >= 2.30      https://git-scm.com/downloads


# ── Firebase (optional — needed for live data & deploy) ──────
Firebase CLI    >= 13.0.0    npm install -g firebase-tools
# Required if you want to:
#   - Deploy Firestore security rules
#   - Deploy Cloud Functions (functions/)
#   - Use Firebase Hosting


# ── Deployment (choose one) ──────────────────────────────────
Vercel CLI      >= 39.0.0    npm install -g vercel
# OR deploy via the Vercel dashboard at https://vercel.com


# ── Environment Variables ────────────────────────────────────
# Create a .env.local file in the project root with:
#
#   VITE_FIREBASE_API_KEY=
#   VITE_FIREBASE_AUTH_DOMAIN=
#   VITE_FIREBASE_PROJECT_ID=
#   VITE_FIREBASE_STORAGE_BUCKET=
#   VITE_FIREBASE_MESSAGING_SENDER_ID=
#   VITE_FIREBASE_APP_ID=
#   VITE_FIREBASE_VAPID_KEY=
#
# Get these from: Firebase Console → Project Settings → Your Apps
# The app works WITHOUT these (uses local demo data as fallback).


# ── Quick Start ───────────────────────────────────────────────
# git clone https://github.com/Nishantghadi14/mandi-minutes-web.git
# cd mandi-minutes-web
# npm install
# npm run dev          → http://localhost:5173
